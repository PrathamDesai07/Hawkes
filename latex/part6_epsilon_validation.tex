\section{The $\eps$-dependence of the regret constant (Problem~6)}\label{sec:epsdep}
%==============================================================

Sources: \texttt{problem06\_epsilon\_explicit.md} (derivations, Sections~1--18), \texttt{code/p6\_*}. Regret is $\mathrm{Reg}_T$ of Definition~\ref{def:regret}; $d=K+1$, $\lbar=\mu/\eps$ is the mean intensity, $n=1-\eps$.

\subsection{Clocks and the target form}\label{sec:eps-clock}
Since $\E\lambda=\mu/\eps$, $n$ events take about $T=\eps n/\mu$ time units, so $\tfrac d2\ln n=\tfrac d2\ln T+\tfrac d2\ln(\mu/\eps)$. A statement $\E R_n=\tfrac d2\ln n+c(\eps)$ in events is the statement $\E R_T=\tfrac d2\ln T+c'(\eps)$ in time with $c=c'-\tfrac d2\ln(\mu/\eps)$. The $C\ln(1/\eps)$ of the target form $\tfrac d2\ln T+C\ln(1/\eps)+\text{burn-in}$ is therefore, for $k_0>1$, the conversion between the two clocks (measured: $\E R_T-\ln T$ is $O(1)$ and flat in $\eps$ in time units, Table~\ref{tab:epsnum}) and not a cost of the algorithm; the one genuine $\ln(1/\eps)$ in time units is $\tfrac{1-k_0}2\ln(1/\eps)$ for $k_0<1$ (Proposition~\ref{prop:k0low}).

\subsection{Three lemmas that remove the event count from the constant}\label{sec:eps-lemmas}

\begin{lemma}[negative moments of the intensity, any $K$]\label{lem:negmom}\Pt{}
Let $Z=\sum_k(\alpha_k/\beta_k)S_k$ (so $\dd Z=-(\lambda-\mu)\dd t+n\dd N$), $b=\min\beta_k$, $B=\max\beta_k$, $r=B/b$, $w=\mu+bZ$ (so $w\le\lambda\le rw$). For $p\ge0$ with $(p+1)Bn^2<2\mu$,
\[
\E_\pi w^{-(p+1)}\le\frac{\eps r}{\mu-(p+1)Bn^2/2}\,\E_\pi w^{-p},
\]
hence $\E_\pi\lambda^{-q}\le\E_\pi w^{-q}<\infty$ exactly for $q<k_{\mathrm{eff}}:=2\mu/(Bn^2)$.
\end{lemma}
\begin{proof}
For $f=w^{-p}$, the alternating Taylor bound $(1+x)^{-p}-1\le-px+p(p+1)x^2/2$ and $1-n=\eps$ give $\cL f\le pb[\eps r w^{-p}-(\mu-(p+1)br n^2/2)w^{-p-1}]$ (using $\lambda\le rw$); $\E_\pi\cL f=0$ ($f\le\mu^{-p}$, $\cL f$ is dominated by polynomial moments). For $K=1$ this is $\E\lambda^{-(p+1)}\le[\gamma/(\beta\mu-(p+1)\alpha^2/2)]\E\lambda^{-p}$, and at $p=0$ Theorem~\ref{thm:A}'s $m\le2\eps/(2\mu-n^2\beta)$.
\end{proof}
Checked against $8$ stationary replicates of $3\cdot10^5$ time units: the bound holds in all $23$ cases where it is finite (ratio bound/truth $1.01$--$36$; for $K=1$, $q=1$ it is within $1$--$9\%$), and it is infinite exactly for $q\ge k_{\mathrm{eff}}$.

\begin{lemma}[leverage-free conversion]\label{lem:defect}\Pt{}
For a PSD increment $\Delta F_i$ and $F=F_{i-1}\succ0$, with $X_i=F^{-1/2}\Delta F_iF^{-1/2}$ (eigenvalues $x_j$),
$\tr X_i=\ln\det(F_i/F_{i-1})+\mathrm{def}_i$, where $\mathrm{def}_i=\sum_j[x_j-\ln(1+x_j)]\le(\tr X_i)^2$.
\end{lemma}
\begin{proof}
$x-\ln(1+x)\le x^2/2$ for $x\le1$ and $\le x\le x^2$ for $x>1$; $\sum x_j^2\le(\sum x_j)^2$.
\end{proof}

\begin{lemma}[gap quantities without a leverage condition]\label{lem:earlyG}\Pt{}
Let $c^\pm$ be the box ratio constants, $\kappa_1=\max(1,1/c^--1)$, $g_i=s_i+h_i$ the gradient (score plus drift part), $M=F_{i-1}^{-1}$. Then $\Delta J_i\preceq c^+\Delta F_i$, $F_{i-1}\succeq F_0+J_{t_{i-1}}/c^+$ pathwise, and
$\E[g_i\T Mg_i\mid\cG_{i-1}]\le2(1+\kappa_1^2)\tr(M\,\E[\Delta J_i\mid\cG_{i-1}])\le2(1+\kappa_1^2)c^+\E[\tr X_i\mid\cG_{i-1}]$.
\end{lemma}
\begin{proof}
$g=s+h$, $g\T Mg\le2s\T Ms+2h\T Mh$; the score identity gives $\E s\T Ms=\tr(M\E\Delta J_i)$; the drift part is rank one, $h=a(1/\lambda^*-1/\lambda_\theta)(\tau)$, so $h\T Mh\le\kappa_1^2a\T Ma/\lambda^{*2}$ and $\E[aa\T/\lambda^{*2}(\tau)\mid\cG_{i-1}]=\E\Delta J_i$.
\end{proof}

\begin{proposition}[decomposition without a bad-gap count]\label{prop:Rdecomp}\Pt{}
Let $t_*$ be deterministic, $N_e=\#\{i:t_{i-1}<t_*\}$ (a stopping time), $L_e=\E\ln\det(F_{N_e}/F_0)$, $D_e=\E\sum_{i\le N_e}\mathrm{def}_i$ and $\Xi_e=2(1+\kappa_1^2)c^+(L_e+D_e)$. Then $\E\sum_{i\le N_e}X_i\le P_0+\Xi_e$, and, with $A=(\tfrac12+C_2\delta)/(1-\delta)$ and the late-gap bounds of Theorem~\ref{thm:R},
\[
\E R_n\le(\tfrac12+C_2\delta)(P_0+\Xi_e+\E G_{\mathrm{late}})+(\tfrac12-C_2\delta)(P_0+\Xi_e)+C_B,
\]
where $G_{\mathrm{late}}$ is the sum over gaps with $t_{i-1}\ge t_*$ and $C_B$ the late bad-gap term.
\end{proposition}
\begin{proof}
Proposition~\ref{prop:reg} stopped at $N_e$ (optional stopping for the square-integrable martingale $M$), Lemma~\ref{lem:earlyG}, $R_i\le X_i$ always and $R_i\le(\tfrac12+C_2\delta)X_i$ on good gaps.
\end{proof}
This replaces the factor $(\mu/\eps)t_*\cdot O(1)$ (number of early gaps times a per-gap bound) of the proof of Theorem~\ref{thm:R} by $P_0+O(c^+d\ln t_*)+O(D_e)$, and removes any high-probability bound on the early parameter error.

\subsection{The potential defect is $O(\eps^a)$ ($K=1$)}\label{sec:eps-defect}

\begin{proposition}[early-phase defect, $K=1$, $k_0>1$, scaled prior]\label{prop:Dearly}\Ps{}
Let $F_0=\rho_0\operatorname{diag}(1/\lbar,\lbar/\beta^2)$, a stationary start, and the information event $\{J_s\succeq\tfrac s2N\ \forall s\ge t_J\}$ with $N$ the tangent minorant (Theorem~\ref{thm:tangent}), and $0<a<(k_{\mathrm{eff}}-1)/2$. Then $D_e\le C_D\eps^a$ with explicit $C_D(a,k_0,\mu,\beta,\rho_0,c^\pm,t_J,\hat C)$.
\end{proposition}
\begin{proof}[Proof (steps)]
(1) $\mathrm{def}_i\le(\tr X_i)^{1+a}$ and $\tr X_i\le(1/c^-)\min\{\tr(F_0^{-1}\Delta J_i),\tr((F_0+J_{t_{i-1}}/c^+)^{-1}\Delta J_i)\}$ (Lemmas~\ref{lem:defect}, \ref{lem:earlyG}): a functional of the true process. (2) For a weight $W$ with $\phi\T W^{-1}\phi\le(C/\omega)\lbar(1+(\beta y/\lbar)^2)$ ($y=Se^{-\beta u}$), $\tr(W^{-1}\Delta J_i)\le(C/\omega)[\lbar T_i+(\beta/\alpha)^2E_i/\lbar]$ with $T_i=\int_0^{\tau_i}\dd u/\lambda^*\le E_i/\lambda_{\mathrm{end}}^2$, $E_i=\Lambda^*(\tau_i)\sim\mathrm{Exp}(1)$ independent of the past; $F_0$-window: $\omega=\rho_0,C=1$; $N$-window: $\omega=t/(2c^+)$, $C=\hat C:=1/(\lbar\lambda_{\min}(SNS))$, $S=\operatorname{diag}(1,\beta/\lbar)$ ($\hat C=6.6$--$7.9$ for $\mu=\beta=1$, $\eps=0.3$--$0.01$: bounded in $\eps$). (3) Given the start intensity $\lambda_0$, $\lambda_{\mathrm{end}}\ge\lambda_0-\beta E$, so $\E[T_i^{1+a}\mid\lambda_0]\le(2/\lambda_0)^{2(1+a)}\Gamma(2+a)+\mu^{-2(1+a)}\Gamma(2+a+m)(2\beta/\lambda_0)^m$. (4) Sampling by events turns sums of pre-event functions into $\int\E[\lambda h(\lambda)]\dd t$, and $\E_{\mathrm{palm}}\lambda_0^{-q}\le B_{q-1}/\lbar$ with $B_q$ the bound of Lemma~\ref{lem:negmom}. (5) Summing: $F_0$-window $\le C't_J(\eps/\mu)^a\rho_0^{-(1+a)}$, $N$-window $\le C'(2c^+\hat C/c^-)^{1+a}(\eps/(\mu t_J))^a/a$, complement of the information event $\le C''e^{-ct_J/2}(\eps/\mu)^a$.
\end{proof}
\emph{Numerical record.} $\E\sum_i\mathrm{def}_i=1.07,0.61,0.17,0.05$ for $\eps=0.3,0.1,0.03,0.01$ against $2\cdot10^4$ to $5.7\cdot10^5$ events (log--log slope $1.09$); $E\,G/\sum\tr X_i=0.99$--$1.11$; $\E X\le P_0+\E G$ in all runs. For a fixed prior the same computation gives $O(\eps^{a-1})$, vacuous for $a<1$ (measured defect $1.1$--$1.4$, flat). For $K\ge2$ the leverage constant $\hat C$ grows like $1/\eps$ ($72,157,461,1329$ for $K=2$, $\eps=0.3,0.1,0.03,0.01$), so the argument needs a mode-adapted bound (\Ot{}); the measured defect is $O(1)$ ($K=2$: $1.03,0.60,0.16$).

\subsection{The $\eps$-explicit theorem}\label{sec:eps-thm}

\begin{theorem}[explicit constants, $K=1$, $k_0>1$]\label{thm:eps}\Ps{}
Let $t_*=\max(t_{\mathrm{lev}},t_{\mathrm{mix}})$, $t_{\mathrm{lev}}=(k_0-1)(1+1/\eta)c^+/(\eps\delta\mu_{\mathrm{lo}})$ (from $\bar\ell(t)=c^+(k_0-1)/(\eps t)$, the supremum at $S\approx0$), $t_{\mathrm{mix}}=c_{\mathrm{mix}}/(\beta\eps\delta^2)$. For $T\ge t_*$, in time units and for the scaled prior,
\[
\E R_T\le A\Big\{P_0+(1+\eta)(1+\delta)^2\big[2\ln T+\ln\det(I/c^-)-\ln\det F_0\big]+(1+\eta)c^+\big[2\ln t_*+\ln\det(I/c^-)-\ln\det F_0\big]+(1+\eta)(D_e+C_\rho)+\tfrac{C_B}{\lambda_{\min}(F_0)}\Big\}+C_B',
\]
with $\ln\det F_0=2\ln\rho_0-2\ln\beta$ (free of $\eps$), $\ln\det I=-2\ln\beta-\ln(k_0-1)+O(\eps^{\min(1,k_0-1/2)})$, $P_0=\rho_0[\eps(\Delta\mu)^2/\mu+(\mu/\eps)(\Delta\alpha/\beta)^2]$, $C_\rho=O(1)$ and $C_B=O(\mu/(c_1\beta\delta^2))$ for the truncation level $p\ln(1/\bar\ell)$ with $p\propto\ln(1/\eps)$ and $t_*\ge t_{\mathrm{mix}}$, and $D_e=O(\eps^a)$ (Proposition~\ref{prop:Dearly}). Leading terms: $\E R_T\le(1+O(\delta+\eta))\{\ln T-\ln\rho_0-\tfrac12\ln(k_0-1)\}+\tfrac12P_0+O(c^+\ln(1/\eps))$.
\end{theorem}
The proof is that of Theorem~\ref{thm:R} with the early window handled by Proposition~\ref{prop:Rdecomp}. The old constant $C=O(\eps^{-2}\ln(1/\eps))$ is a proof artefact (the event count); the additive constant is $\eps$-flat in time units.

\begin{proposition}[the ridge in $\det I$, $k_0\le1$]\label{prop:k0low}\Pt{}/\Nt{}
$\det I=(mE\lambda-1)/\alpha^2$, $E\lambda=\mu/\eps$, so Theorem~\ref{thm:A} gives $\det I\to1/(\beta^2(k_0-1))$ ($k_0>1$), $\det I\approx(\ln(1/\eps)-1)/\beta^2$ ($k_0=1$), $\det I\approx(C\eps^{k_0-1}-1)/\beta^2$ ($k_0<1$): for $k_0<1$ the regret constant contains $\tfrac12\ln\det I\approx\tfrac{1-k_0}2\ln(1/\eps)$. Measured ($\beta=1$, $\eps=0.3\to0.003$): $\mu=0.3$: $\det I=1.50,2.70,5.24,9.20,15.3$, fit $1.5\eps^{-0.4}-1$ (the implied $C$: $1.54,1.47,1.54,1.62,1.57$); $\mu=0.5$: $1.00,1.57,2.45,3.40,4.41$ against $\ln(1/\eps)-1=0.20,1.30,2.51,3.61,4.81$.
\end{proposition}

\paragraph{Burn-in (measured).} The exit time $T_{\mathrm{exit}}(\delta)$ from $R(\delta)$ obeys $T_{\mathrm{exit}}\approx z^2\max_j(I^{-1})_{jj}/(\delta\theta^*_j)^2$ with $z^2=1.4$ (observed/predicted $0.85$--$1.23$ in all ten runs of Section~\ref{sec:sim}), i.e. $\approx30/\eps$ for $K=1$, $\mu=\beta=1$, and for $K=2$ it is set by the middle tier ($\eps$-independent, $500$--$1100$). The earlier reading that the burn-in is mixing-limited and $\mu$-independent is superseded by this information form (they coincide for $\mu=\beta=1$). The start cost is exactly $\tfrac12P_0$ for a prior scaled to the information level ($12.5$ measured against $12.0$ predicted at $\eps=0.01$).

\section{Numerical validation: simulation study, lower-bound numerics, baselines, real data (Problems 8, 9, 19--21)}\label{sec:validation}

\subsection{The regret notion and the lower bound, checked}\label{sec:val-reg}
\emph{Regret notion (Section~\ref{sec:regretnotion}).} $\E r_T-\E R_T$ is within one standard error in 4000-replicate runs ($-0.013\pm0.018$, $-0.018\pm0.026$ for $K=1$, $\eps=0.1$; $K=2$: $-0.005\pm0.018$) and within $2$ standard errors in $28$ single-gap cells (exact sampling); Lemma~\ref{lem:regsand} has no violation over $\approx2\cdot10^6$ gaps per run (violations at $R,Q\sim10^{-13}$ were rounding); $\sum R/(\sum Q/2)=0.95$--$1.01$; $\norm{\bar\theta_n-\theta^*}^2_{J_n}\le2\tilde c^+R_n$ holds with margin $\ge1.4$; $\E D_n=1.9$--$2.0$ (flat in $n$) while $\E R_n$ grows; the hindsight regret exceeds $\mathrm{Reg}_n$ by $1.0$ ($K=1$) and $1.4$ ($K=2$) against $d/2=1,1.5$ (Wilks).

\emph{Lower bound (Proposition~\ref{prop:LBmat}, Theorem~\ref{thm:LBeps}).} Over the stability band $\Theta_0(\eps)=[\mu(1\pm w)]\times[\beta(1-\tfrac32\eps),\beta(1-\tfrac12\eps)]$ ($w=0.25$), $60$--$300$ replicates:
(i) the bound with the \emph{measured} covariance, $L_{\mathrm{emp}}=\ln|\Theta_0|-\tfrac12\ln((2\pi e)^d\det\mathrm{Cov})$, satisfies $L_{\mathrm{emp}}\le\E R_T$ (Newton) in all $31$ rows, with gap $0.05$--$0.46$; (ii) \emph{scaling collapse}: the Bayes regret depends on $x=T\eps$ only, $\E R=0.165\pm0.01$ at $x=3$ and $0.79$--$0.90$ at $x=30$ for $\eps$ from $0.3$ to $0.003$ (a factor $100$); (iii) $\E R\approx I_G(T)-0.35$ at large $x$, with $I_G=\tfrac12\ln\det(I_d+T\Sigma^{1/2}I\Sigma^{1/2})$ and $0.35=\tfrac d2\ln(2\pi e/12)$ the uniform-versus-Gaussian prior entropy; (iv) with $A=N$ (tangent minorant) the form of Theorem~\ref{thm:LBeps} gives $\E R-L_N=0.78,0.83,0.57,1.10,1.00$ for $\eps=0.3,\dots,0.003$; (v) the efficiency constant is $\kappa_{\mathrm{eff}}\approx1$ ($0.8$--$1.26$ for $T\eps\ge30$), so the $\tfrac d2\ln\ln T$ of the theorem is a proof artefact; (vi) the additive constant $c=\E R-\ln T-\tfrac12\ln\det\bar I-\ln|\Theta_0|$ is flat in $\eps$ and in $\mu$ ($-1.6$ to $-2.0$ at $T\eps=30$, including $k_0=0.6$, where $\tfrac12\ln\det\bar I=0.59,0.94$ is needed). A predicted $(d-1)/2$ plateau at $T\eps<1$ is \emph{wrong} for this band. Proposition~\ref{prop:twopoint} (two-point KL, Fisher form): ratio $\E R_T/(Tu\T Iu/2)=0.965$--$1.000$ ($\delta\le0.4\eps$; $0.993$--$1.000$ at $\delta=0.1\eps$), always inside $[1/\tilde c^+,1/\tilde c^{-2}]$.

\subsection{Real data: Deribit BTC option trades (Problems 20 and 21)}\label{sec:real}
\emph{Data.} NSE/Bank Nifty tick data is not available in the working environment (the Bank Nifty file is $1$-minute option-chain snapshots, from which event times cannot be recovered). The tick data used are $5{,}411{,}992$ Deribit BTC option trades, 2024-11-01 to 2025-10-31, millisecond timestamps; an event is a distinct millisecond ($4{,}770{,}908$ events, $11.8\%$ of trades collapsed); window $13$--$20$ UTC daily, $365$ windows, median $4{,}435$ events.
\emph{Dictionary.} Held-out log-likelihood improves monotonically with $K$ and with the time-scale span (validation days $30$--$59$: $-2.309$ for $\beta=0.03$ to $-1.982$ for $(0.1,1,10,100)$); the single-exponential $\hat n$ ranges over $0.21$--$0.91$ with $\beta$. Main dictionary $\beta=(0.1,1,10)/$s.
\emph{Results.} Daily batch $\hat n$: median $0.668$ (10--90\% $0.561$--$0.768$, range $0.44$--$0.93$), dispersion $5.0\times$ its standard error ($0.017$), $\mathrm{corr}(\hat n,\log\text{events})=+0.77$, weekends $0.596$ against weekdays $0.684$; the two largest bursts differ in kind (2025-03-02: $\hat n=0.93$ with low $\hat\mu$; 2025-10-10: $\hat\mu=0.118$, $\hat n=0.80$). Misspecification: median $\hat n=0.50,0.55,0.67,0.66$ for $K=1,2,3,4$, the time-rescaling test rejects $K\le3$ on $98$--$100\%$ of days and $K=4$ (with the $100/$s component) on $15\%$, so the standard errors for $K\le3$ are optimistic; the bias-versus-$K$ picture of Section~\ref{sec:mis-bias} is already saturated at $K=3$. The growing-window online estimate equals the pooled batch MLE to $0.0025$ ($0.7565$ against $0.7590$ after $365$ days); the offset to the daily median ($+0.04$ to $+0.09$) is regime heterogeneity (event-weighted mean of the daily values $0.697$), not estimator error.

\paragraph{Scope of the evidence.} One stream (crypto, not NSE), one dictionary, one development/test split; the Hall--Willett and Solo estimators are adaptations; the intervals are day-block bootstraps and do not cover hyperparameter selection or regime uncertainty; the log-loss gain is a predictive statement about arrival times and carries no trading claim.
